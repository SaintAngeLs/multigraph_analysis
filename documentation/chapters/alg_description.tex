\section{Opis algorytmów}

\subsection{Algorytm obliczania rozmiaru multigrafu}
\begin{algorithm}[H]
\caption{Obliczanie rozmiaru multigrafu}
\begin{algorithmic}[1]
\Require \( G = (V, E) \) - multigraf reprezentowany przez macierz sąsiedztwa \( A \)
\Ensure Rozmiar multigrafu \( |E| \)
\State \( |E| \gets 0 \)
\ForEach{wierzchołki \( i, j \) w \( V \)}
    \State \( |E| \gets |E| + A[i][j] \)
\EndFor
\State \Return \( |E| \)
\end{algorithmic}
\end{algorithm}

Złożoność obliczeniowa algorytmu wynosi \( O(n^2) \), gdzie \( n \) to liczba wierzchołków w multigrafie. Wynika to z iteracji po całej macierzy sąsiedztwa \( A \) o wymiarach \( n \times n \).

\subsection{Algorytm wyszukiwania cykli w multigrafie}
\begin{algorithm}[H]
\caption{Znajdowanie cykli w multigrafie}
\begin{algorithmic}[1]
\Require \( G = (V, E) \) - multigraf reprezentowany przez macierz sąsiedztwa \( A \)
\Ensure Liczba cykli w \( G \)
\State \( \text{cycle\_count} \gets 0 \)
\State Utwórz pusty zbiór \( \text{visited} \)
\Function{DFS}{( u, start, path )}
    \If{\( \text{length}(path) > 2 \textbf{ and } u = \text{start} \)}
        \State \( \text{cycle\_count} \gets \text{cycle\_count} + 1 \)
        \State \Return
    \EndIf
    \ForEach{\( v \) sąsiadujący z \( u \)}
        \If{\( v \notin \text{visited} \)}
            \State Dodaj \( v \) do \( \text{visited} \)
            \State \Call{DFS}{( v, start, path + [v] )}
            \State Usuń \( v \) z \( \text{visited} \)
        \EndIf
    \EndFor
\EndFunction
\ForEach{wierzchołek \( u \in V \)}
    \State \( \text{visited} \gets \{u\} \)
    \State \Call{DFS}{( u, u, [u] )}
\EndFor
\State \Return \( \text{cycle\_count} \)
\end{algorithmic}
\end{algorithm}

\subsection{Algorytm liczenia cykli Hamiltona}
\begin{algorithm}[H]
\caption{Liczenie cykli Hamiltona w multigrafie}
\begin{algorithmic}[1]
\Require \( G = (V, E) \) - multigraf reprezentowany przez macierz sąsiedztwa \( A \)
\Ensure Liczba cykli Hamiltona
\State \( \text{count} \gets 0 \)
\Function{HamiltonianBacktrack}{( u, visited, depth )}
    \If{\( \text{depth} = |V| \textbf{ and } A[u][\text{start}] > 0 \)}
        \State \( \text{count} \gets \text{count} + 1 \)
        \State \Return
    \EndIf
    \ForEach{\( v \) sąsiadujący z \( u \)}
        \If{\( v \notin \text{visited} \)}
            \State Dodaj \( v \) do \( \text{visited} \)
            \State \Call{HamiltonianBacktrack}{( v, visited, depth + 1 )}
            \State Usuń \( v \) z \( \text{visited} \)
        \EndIf
    \EndFor
\EndFunction
\ForEach{wierzchołek \( u \in V \)}
    \State \( \text{visited} \gets \{u\} \)
    \State \Call{HamiltonianBacktrack}{( u, \text{visited}, 1 )}
\EndFor
\State \Return \( \text{count} \)
\end{algorithmic}
\end{algorithm}

\subsection{Algorytm obliczania metryki pomiędzy dwoma multigrafami}

\begin{algorithm}
\caption{Obliczanie rozkładu stopni wierzchołków}
\label{alg:degree_distribution}
\begin{algorithmic}[1]
\Require Macierz sąsiedztwa \textit{adj\_matrix}, Liczba wierzchołków $n$
\Ensure Rozkład stopni wierzchołków \textit{degree\_count}

\State Inicjalizacja pustej tablicy z haszowaniem \textit{degree\_count}
\ForEach{wierzchołek $i$ w grafie}
\State Set \textit{degree} $\gets 0$
\ForEach{sąsiad $j$ wierzchołka $i$}
\State \( \textit{degree} \gets \text{degree} + \textit{adj\_matrix[$i$][$j$]}\)
\EndFor
\If{\textit{degree} nie występuje w  \textit{degree\_count}}
\State Dodaj \textit{degree} do \textit{degree\_count} jako klucz z wartością 1
\Else
\State Inkrementuj wartość dla klucza \textit{degree} w \textit{degree\_count}
\EndIf
\EndFor

\State Znormalizuj \textit{degree\_count} dzieląc każdą wartość przez $n$
\Return \textit{degree\_count}
\end{algorithmic}
\end{algorithm}

\begin{algorithm}
\caption{Oblicz Earth Mover's Distance (EMD)}
\label{alg:calculate_emd}
\begin{algorithmic}[1]
\Require Rozkłady stopni wierzchołków dla dwóch grafów \textit{dist1}, \textit{dist2}
\Ensure Wartość EMD

\State Inicjalizacja pustej listy \textit{all\_degrees}
\State Znalezienie sumy zbiorów kluczy \textit{dist1} and \textit{dist2} jako \textit{all\_degrees}
\State Posortowanie rosnąco \textit{all\_degrees}

\State \textit{cumulative\_dist1} $\gets 0$ i \textit{cumulative\_dist2} $\gets 0$
\State \textit{emd} $\gets 0$

\ForEach{degree in \textit{all\_degrees}}
\State \textit{prob1} $\gets  \textit{dist1}[degree]$, \textit{prob2} $\gets \textit{dist2}[degree]$ ($0$ w przypadku braku klucza)
\State \textit{cumulative\_dist1} $\gets$ \textit{cumulative\_dist1} $+$ \textit{prob1}
\State \textit{cumulative\_dist2} $\gets$ \textit{cumulative\_dist2} $+$ \textit{prob2}
\State \textit{emd} $\gets \textit{emd} + \left| \textit{cumulative\_dist1} - \textit{cumulative\_dist2} \right|$
\EndFor
\State \Return \textit{emd}

\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Obliczanie metryki pomiędzy dwoma multigrafami}
\begin{algorithmic}[1]
\Require \( G = (V_G, E_G) \) oraz \( H = (V_H, E_H) \) - dwa multigrafy reprezentowane przez macierze sąsiedztwa \( A_G \) i \( A_H \)
\Ensure Metryka pomiędzy \( G \) i \( H \)
\State $dist1 \gets \textsc{CalculateDegreeDistribution}(A_G, \left| V_G \right|)$
\State $dist2 \gets \textsc{CalculateDegreeDistribution}(A_H, \left| V_H \right|)$
\State $metric \gets \textsc{CalculateEMD}(dist1, dist2)$
\State \Return \( \text{metric} \)
\end{algorithmic}
\end{algorithm}



\subsection{Algorytm minimalnego rozszerzenia dla cyklu Hamiltona}
\begin{algorithm}[H]
\caption{Minimalne rozszerzenie dla cyklu Hamiltona}
\begin{algorithmic}[1]
\Require \( G = (V, E) \) - multigraf reprezentowany przez macierz sąsiedztwa \( A \)
\Ensure Minimalna liczba krawędzi do dodania
\State \( \text{min\_edges} \gets \infty \)
\Function{DFS}{( u, visited, added\_edges )}
    \If{\( \text{length}(visited) = |V| \textbf{ and } A[u][\text{start}] > 0 \)}
        \State \( \text{min\_edges} \gets \min(\text{min\_edges}, \text{added\_edges}) \)
        \State \Return
    \EndIf
    \ForEach{\( v \) sąsiadujący z \( u \)}
        \State Dodaj \( v \) do \( \text{visited} \)
        \If{\( A[u][v] = 0 \)}
            \State Dodaj krawędź \( (u, v) \)
            \State \Call{DFS}{( v, visited, added\_edges + 1 )}
            \State Usuń krawędź \( (u, v) \)
        \Else
            \State \Call{DFS}{( v, visited, added\_edges )}
        \EndIf
        \State Usuń \( v \) z \( \text{visited} \)
    \EndFor
\EndFunction
\ForEach{wierzchołek \( u \in V \)}
    \State \( \text{visited} \gets \{u\} \)
    \State \Call{DFS}{( u, \text{visited}, 0 )}
\EndFor
\State \Return \( \text{min\_edges} \)
\end{algorithmic}
\end{algorithm}
